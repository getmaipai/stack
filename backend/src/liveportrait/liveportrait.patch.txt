+# coding: utf-8
+"""Stack patch (TALK-06): the licence-clean face finder.
+
+LivePortrait ships a face finder whose model files are non-commercial research
+only. The Stack never installs them. This finder takes a face rectangle from a
+detector the Stack supplies (Apple Vision through a small compiled helper, then
+MediaPipe, both permissive) and turns it into the 203 landmarks that
+`landmark.onnx` (MIT) predicts on a crop around the rectangle. The cropper then
+refines them exactly as it refines landmarks from any other source.
+"""
+from types import SimpleNamespace
+
+import cv2
+import numpy as np
+
+from .crop import _transform_pts, crop_image_by_bbox
+
+# Set by the Stack's worker: a callable taking an RGB array and returning (x, y, w, h) in pixels, or None.
+detect_face = None
+# The landmark model sees a crop this many times the rectangle's longer side.
+CROP_SCALE = 1.5
+
+
+class FaceFinder(object):
+    def __init__(self, landmark_runner):
+        self.landmark_runner = landmark_runner
+
+    def get(self, img_bgr, flag_do_landmark_2d_106=True, direction="large-small", max_face_num=0):
+        """Same shape of answer as the finder it replaces: a list of faces, each with `landmark_2d_106`.
+        The array holds 203 points (the cropper accepts either count)."""
+        if detect_face is None:
+            return []
+        img_rgb = np.ascontiguousarray(img_bgr[..., ::-1])
+        box = detect_face(img_rgb)
+        if box is None:
+            return []
+        x, y, w, h = box
+        side = max(w, h) * CROP_SCALE
+        cx, cy = x + w / 2.0, y + h / 2.0
+        dsize = self.landmark_runner.dsize
+        crop = crop_image_by_bbox(img_rgb, [cx - side / 2, cy - side / 2, cx + side / 2, cy + side / 2], dsize=dsize, borderMode=cv2.BORDER_CONSTANT)
+        inp = (crop["img_crop"].astype(np.float32) / 255.0).transpose(2, 0, 1)[None, ...]
+        out = self.landmark_runner._run(inp)
+        lmk = np.asarray(out[2][0]).reshape(-1, 2) * dsize
+        return [SimpleNamespace(landmark_2d_106=_transform_pts(lmk, crop["M_c2o"]))]
